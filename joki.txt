    <script>
        function toggleMenu() {
            document.getElementById('mobileMenu').classList.toggle('terbuka');
        }

        function filterProduk(btn, kategori) {
            document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('aktif'));
            btn.classList.add('aktif');

            const kartu = document.querySelectorAll('#gridProduk .kartu-produk');
            kartu.forEach(k => {
                if (kategori === 'semua') {
                    k.style.display = 'flex';
                } else {
                    const kats = k.dataset.kategori || '';
                    k.style.display = kats.includes(kategori) ? 'flex' : 'none';
                }
            });
        }
    </script>
